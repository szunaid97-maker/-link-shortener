const db = getFirestore(app);

async function shortenUrl() {
    const longUrl = document.getElementById('longUrl').value;
    if (!longUrl) {
        alert('Please enter a long link');
        return;
    }
    
    const shortId = Math.random().toString(36).substring(7);
    const shortUrl = "[https://jr-link-shortener.talk4fun.net/$](https://jr-link-shortener.talk4fun.net/$){shortId}";
    
    try {
        await setDoc(doc(db, "links", shortId), {
            longUrl: longUrl,
            shortUrl: shortUrl,
            createdAt: new Date()
        });
        
        document.getElementById('result').innerHTML = Shortened link: <a href="${shortUrl}" target="_blank">${shortUrl}</a>;
    } catch (e) {
        console.error("Error adding document: ", e);
        alert("Failed to shorten link.");
    }
}

window.shortenUrl = shortenUrl;
